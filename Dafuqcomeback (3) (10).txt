                local status = char:FindFirstChild("Status")

                        isCharging = true

                        local last = current
                        local noUp = 0

                        for i = 1,8 do
                            task.wait(0.5)

                            local now = energy.Value
                            if now <= last + 20 then
                                noUp += 1
                                if noUp >= 4 then
                                    break
                                end
                            else
                                noUp = 0
                                last = now
                            end
                        end

                        game.ReplicatedStorage.Remotes.SkillRemote:FireServer(unpack(releaseArgs))
                        isCharging = false
                    end
                end

                task.wait(0.8)
            end
        end)

            Text = Text.."CritDamage : "..tostring(CritDamage.Value).."\n"

        for _,v in ipairs(Status:GetChildren()) do
            if v.Name ~= "CritChance" and v.Name ~= "CritDamage" then
                if v:IsA("NumberValue")
                or v:IsA("IntValue")
                or v:IsA("BoolValue")
                or v:IsA("StringValue") then

                    Text = Text..v.Name.." : "..tostring(v.Value).."\n"
                end
            end
        end

        OrionLib:MakeNotification({
            Name = SelectedPlayer.." Status",
            Content = Text,
            Time = 10
        })
    end

local function GetMobList()
    local List = {}
    for _, v in ipairs(BossFolder:GetChildren()) do
        table.insert(List, v.Name)
    end
    for _, v in ipairs(MobFolder:GetChildren()) do
        if not table.find(List, v.Name) then
            table.insert(List, v.Name)
        end

    table.sort(List)
    return List

	if IsRecording and CurrentRecording then

local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    if IsRecording and method == "FireServer" then
        local class = self.ClassName
        if class == "RemoteEvent" or class == "RemoteFunction" then
            RecordAction({
                Type = "Remote",
                Remote = self,
                Args = {...},
            })
        end
    end
    return oldNamecall(self, ...)
end)

local function GetRoot(model)
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("PrimaryPart")
        or model:FindFirstChild("Torso")
end

local function GetNearestMob()
    local character = LP.Character
    if not character then return nil end

    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    local nearest
    local dist = Settings.Range

    local folders = {}

    -- Mobs
    if Settings.TargetMobs then
        local folder = workspace["World Mobs"]:FindFirstChild("Mobs")
        if folder then table.insert(folders, folder) end
    end

    -- Boss
    if Settings.TargetBoss then
        local folder = workspace["World Mobs"]:FindFirstChild("Boss Mobs")
        if folder then table.insert(folders, folder) end
    end

    -- Boss Event
    if Settings.TargetEventBoss then
        local folder = workspace["World Mobs"]:FindFirstChild("Event Mobs")
        if folder then table.insert(folders, folder) end
    end

    for _, folder in ipairs(folders) do
        for _, mob in ipairs(folder:GetChildren()) do
            local root = GetRoot(mob)
            local hum = mob:FindFirstChildWhichIsA("Humanoid")

            if root and hum and hum.Health > 0 then
                local d = (root.Position - hrp.Position).Magnitude
                if d < dist then
                    dist = d
                    nearest = mob
                end
            end
        end
    end

    return nearest
end

local DEAD_TIME_REQUIRED = 15
local CHECK_INTERVAL = 0.5
local isEnabled = false
local deadStartTime = nil

-- Kiểm tra còn Event Mobs
local function hasEventMobs()
	local worldMobs = workspace:FindFirstChild("World Mobs")
	if not worldMobs then return false end

	local eventMobs = worldMobs:FindFirstChild("Event Mobs")
	if not eventMobs then return false end

	for _, mob in ipairs(eventMobs:GetChildren()) do
		if mob:IsA("Model") or mob:IsA("BasePart") then
			local hum = mob:FindFirstChildOfClass("Humanoid")
			if hum then
				if hum.Health > 0 then return true end
			else
				return true
			end
		end
	end
	return false
end

-- Gọi StartDungeon
local function triggerStartDungeon()
	pcall(function()
		local packages = ReplicatedStorage:WaitForChild("Packages")
		local index = packages:WaitForChild("_Index")
		local knitPackage = index:WaitForChild("sleitnick_knit@1.4.7")
		local knit = knitPackage:WaitForChild("knit")
		local services = knit:WaitForChild("Services")
		local dungeonService = services:WaitForChild("DungeonLobbyService")
		local rf = dungeonService:WaitForChild("RF")
		local startDungeon = rf:WaitForChild("StartDungeon")
		startDungeon:InvokeServer()
	end)
end

-- Character
player.CharacterAdded:Connect(function(char)
	character = char
	humanoid = char:WaitForChild("Humanoid")
	deadStartTime = nil
end)

-- Loop
task.spawn(function()
	while true do
		task.wait(CHECK_INTERVAL)

		if not isEnabled then
			deadStartTime = nil
			continue
		end

		if not humanoid or not humanoid.Parent then continue end

		if humanoid.Health <= 0 then
			if not deadStartTime then
				deadStartTime = tick()
			elseif tick() - deadStartTime >= DEAD_TIME_REQUIRED then
				if hasEventMobs() then
					triggerStartDungeon()
					deadStartTime = nil
					task.wait(3)
				end
			end
		else
			deadStartTime = nil
		end
	end
end)
